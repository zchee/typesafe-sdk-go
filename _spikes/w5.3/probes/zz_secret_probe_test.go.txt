//go:build !race

package typesafe

import (
	"net/http"
	"testing"
)

func TestZZSecretProbe(t *testing.T) {
	for _, n := range []string{"Content-Type", "X-Typesafe-Request-Id", "Authorization", "X-Access-Token", "set-cookie", "Date"} {
		t.Logf("isSecretHeader(%q) allocs=%v", n, testing.AllocsPerRun(200, func() { _ = isSecretHeader(n) }))
	}
	h := http.Header{"Content-Type": {"application/json"}, "Date": {"x"}, "X-Typesafe-Request-Id": {"req_1"}, "Set-Cookie": {"a=b"}}
	var sink http.Header
	t.Logf("headerRedactor{}.header(4 headers) allocs=%v", testing.AllocsPerRun(200, func() { sink = headerRedactor{}.header(h) }))
	t.Logf("requestCredentials(4 headers) allocs=%v", testing.AllocsPerRun(200, func() { _ = requestCredentials(h) }))
	_ = sink
}
